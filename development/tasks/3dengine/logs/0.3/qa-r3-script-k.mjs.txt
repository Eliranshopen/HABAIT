// qa-r3 k.mjs: real kills at random moments/events + output damage (dead and alive) + cache damage.  usage: node k.mjs <seed> <jobs>
import fs from 'node:fs';
import { mkEnv, rng, inv, overlaps, setEnv, clearEnv, finish, resume, sleep, pidAlive, STAGES, PAID, settledSum } from './q.mjs';
const seed = Number(process.argv[2] || 1); const N = Number(process.argv[3] || 10);
const R = rng(seed);
const E = await mkEnv({ prefix: 'qa3k-', callUsd: 0.6, stepMs: 300, heartbeat: 0.1, staleSeconds: 1 });
let kills = 0; const viol = []; const notes = [];
const v = (a) => { for (const x of a) { viol.push(x); console.log('VIOLATION', x); } };

const events = {
  delay: (id) => Date.now() - t0 > D,
  joblock: (id) => fs.existsSync(E.jobPath(id, 'job.lock')),
  tmp: (id) => fs.readdirSync(E.jobPath(id, '')).some((f) => f.endsWith('.tmp')) || (fs.existsSync(E.jobPath(id, 'cache')) && fs.readdirSync(E.jobPath(id, 'cache')).some((f) => f.endsWith('.tmp'))),
  reserved: (id) => E.ledger(id).rows.some((r) => r.status === 'reserved'),
  cacheUnsettled: (id) => E.ledger(id).rows.some((r) => r.status === 'reserved' && fs.existsSync(E.jobPath(id, `cache/${r.call_key}.json`))),
  heavy: () => fs.existsSync(E.config.runner.heavyLock),
  heavyguard: () => fs.existsSync(`${E.config.runner.heavyLock}.guard`),
  approval: (id) => E.job(id).state === 'awaiting_approval',
  state: (id) => E.job(id).state === S,
  rowsn: (id) => E.rows(id).length >= RN,
};
let t0, D, S, RN;
async function killOn(id, ev, timeout = 8000) {
  t0 = Date.now();
  const until = t0 + timeout;
  while (Date.now() < until) {
    try { if (E.job(id).state === 'done') return false; if (events[ev](id)) { const p = E.runnerPid(id); if (p && pidAlive(p)) { process.kill(p); for (let i = 0; i < 200 && pidAlive(p); i++) await sleep(5); kills++; return true; } } } catch { /* file in flux */ }
    if (E.job(id).state === 'awaiting_approval' && !E.job(id).approval && ev !== 'approval') E.approve(id);
    await sleep(1);
  }
  return false;
}
function damage(id, mode) {
  const picks = [];
  const stg = STAGES.filter((s) => fs.existsSync(E.jobPath(id, `stages/${s}/stub.json`)));
  if (!stg.length) return picks;
  const n = mode === 'multi' ? R.int(2, Math.min(4, stg.length)) : 1;
  const chosen = [...stg].sort(() => R() - 0.5).slice(0, n);
  for (const s of chosen) {
    const f = E.jobPath(id, `stages/${s}/stub.json`);
    const how = R.pick(['delete', 'garbage', 'truncate', 'flip']);
    if (how === 'delete') fs.rmSync(f); else if (how === 'garbage') fs.writeFileSync(f, 'damaged\n'); else if (how === 'truncate') fs.writeFileSync(f, ''); else fs.writeFileSync(f, fs.readFileSync(f, 'utf8').replace(/"/, ' "'));
    picks.push(`${s}:${how}`);
  }
  return picks;
}
const evNames = ['delay', 'joblock', 'tmp', 'reserved', 'cacheUnsettled', 'heavy', 'heavyguard', 'approval', 'state', 'rowsn'];
for (let i = 0; i < N; i++) {
  clearEnv();
  const name = `k${seed}-${i}.pdf`;
  const pdf = E.pdf(name, name);
  const hooks = R() < 0.5 ? { [R.pick(['ENGINE_TEST_PAUSE_AFTER_RESERVE_MS', 'ENGINE_TEST_PAUSE_AFTER_CACHE_MS', 'ENGINE_TEST_PAUSE_AFTER_SETTLE_MS'])]: R.int(300, 1500) } : {};
  setEnv(hooks);
  const id = E.h.start_simulation({ pdf_path: pdf }).job_id;
  clearEnv();
  const plan = [];
  try {
    const nk = R.int(1, 3);
    for (let k = 0; k < nk; k++) {
      const ev = R.pick(evNames); D = R.int(0, 1800); S = R.pick(STAGES); RN = R.int(1, 7);
      const got = await killOn(id, ev);
      plan.push(`${ev}${got ? '*' : '-'}`);
      if (!got) break;
      v(inv(E, id, { quiet: false, tag: `[k${i} after kill ${ev}]` }));
      // damage while DEAD (sometimes)
      let dm = [];
      if (R() < 0.5) {
        const before = E.rows(id).filter((r) => !r.superseded);
        const ledN = E.ledger(id).rows.length;
        dm = damage(id, R() < 0.5 ? 'multi' : 'one');
        plan.push(`dmg(${dm.join(',')})`);
        if (dm.length) notes.push({ id, dm, ledN, att: { ...E.job(id).attempts } });
      }
      await resume(E, pdf);
      await sleep(R.int(0, 400));
    }
    // damage while ALIVE (sometimes): then let it finish
    if (R() < 0.35) { await sleep(R.int(100, 600)); const dm = damage(id, 'one'); plan.push(`ALIVEdmg(${dm.join(',')})`); }
    const j0 = E.job(id);
    if (j0.state === 'cancelled' || j0.state === 'failed') await resume(E, pdf);
    if (!pidAlive(E.runnerPid(id) || 0)) await resume(E, pdf);
    await finish(E, id, 90000);
    await E.waitFor('runner gone', () => E.runnerPid(id) === null, 15000, 20);
    const iv = inv(E, id, { quiet: true, expectDone: true, tag: `[k${i} ${plan.join('>')}]` });
    // damage never makes a paid stage bill again (cache intact): every paid stage sent exactly once unless orphan-resend
    for (const st of PAID) {
      const led = E.ledger(id).rows.filter((r) => r.stage === st);
      const orph = led.filter((r) => r.status === 'orphaned').length;
      if (E.sends(id, st) > 1 + orph) iv.push(`[k${i}] ${st} sent ${E.sends(id, st)}x with ${orph} orphan rows`);
    }
    v(iv);
    console.log(`job ${i} ok-or-viol plan=${plan.join('>')} attempts=${JSON.stringify(E.job(id).attempts)} spent=${E.job(id).cost.spent_usd}`);
  } catch (e) { viol.push(`[job ${i}] EXC ${e.message}`); console.log('EXC', i, plan.join('>'), e.message); }
}
const ids = fs.readdirSync(E.config.jobsDir);
v(overlaps(E, ids));
console.log(`SUMMARY seed=${seed} jobs=${N} kills=${kills} violations=${viol.length}`);
await E.cleanup();
console.log('leftover pids:', JSON.stringify(E.leftover));
process.exit(viol.length ? 1 : 0);
