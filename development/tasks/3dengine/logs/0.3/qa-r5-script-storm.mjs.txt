import fs from 'node:fs';
import { createHash } from 'node:crypto';
import { spawn } from 'node:child_process';
import { lib, check, rec, sleep, pidAlive, counters, procM } from './h.mjs';

const [suite, seedArg, countArg] = [process.argv[2], Number(process.argv[3] || 1), Number(process.argv[4] || 5)];
const C = counters();
let s = seedArg >>> 0;
const rnd = () => { s = (s * 1664525 + 1013904223) >>> 0; return s / 2 ** 32; };
const pick = (a) => a[Math.floor(rnd() * a.length)];
const sha = (f) => createHash('sha256').update(fs.readFileSync(f)).digest('hex');
const PAID = ['skinning', 'gating', 'finishing'];
const STAGES = ['tracing', 'building', 'rendering', 'skinning', 'gating', 'finishing', 'packaging', 'qa'];
const setHooks = (o) => { for (const [k, v] of Object.entries(o)) process.env[k] = String(v); };
const clearHooks = () => { for (const k of [...lib.HOOK_VARS, ...lib.FREEZE_VARS]) delete process.env[k]; };
const approveWhenAsked = (E, id) => { const j = E.job(id); if (j.state === 'awaiting_approval' && !j.approval) E.approve(id); return j; };
const toDone = (E, id, ms = 60000) => E.waitFor(`done ${id}`, () => { const j = approveWhenAsked(E, id); if (j.state === 'failed') throw new Error('failed ' + JSON.stringify(j.reason)); return j.state === 'done'; }, ms, 20);
async function spawnMany(E, id, n) { const pids = []; for (let i = 0; i < n; i++) { const p = spawnRunnerDefault(E.config, id); E.spawned.add(p); pids.push(p); } return pids; }
async function killRunner(E, id) { const p = await E.kill(id); C.add('kills'); return p; }
const settle = async (E, id) => { await E.waitFor('quiet', () => { const p = E.runnerPid(id); return !p || !pidAlive(p); }, 15000, 50).catch(() => {}); await sleep(300); };

// A: killed inside keep() while the answer is in hand and the job lock is busy; three runners resume at once.
async function A(E, i) {
  const secs = pick([3, 6]);
  const pdf = E.pdf(`A${i}.pdf`, `A-${seedArg}-${i}`);
  const wake = `${E.home}/wakeA${i}`;
  const point = `${pick(PAID)}:sent`;
  setHooks({ ENGINE_TEST_FREEZE_MS: 120000, ENGINE_TEST_FREEZE_AT: point, ENGINE_TEST_FREEZE_WAKE: wake });
  let id; try { id = E.h.start_simulation({ pdf_path: pdf }).job_id; } finally { clearHooks(); }
  const pid = await E.waitFor('pid', () => E.runnerPid(id), 15000);
  await E.waitFor('blocked', () => { approveWhenAsked(E, id); return E.log(id).includes(`at ${point} `) && E.log(id).includes(`runner ${pid} test hook: thread blocked`); }, 45000);
  const lockFile = E.jobPath(id, 'job.lock');
  const h = spawn(process.execPath, [`${process.env.QA_DIR}/holder.mjs`, lockFile, String(secs), pick(['real', 'plant', 'refresh'])], { stdio: 'ignore', windowsHide: true });
  await E.waitFor('held', () => fs.existsSync(lockFile + '.held'), 8000, 10);
  fs.writeFileSync(wake, 'w');
  await E.waitFor('keep waits', () => /waits \(/.test(E.log(id)), 20000, 40);
  await killRunner(E, id);
  await E.waitFor('holder gone', () => !pidAlive(h.pid), 20000, 100);
  fs.rmSync(lockFile, { force: true });
  await spawnMany(E, id, 3);
  await toDone(E, id); await settle(E, id);
  const v = check(E, id, { label: `A${i}[${point}]` });
  rec(`A keep-kill ${point} hold=${secs}`, v.length === 0, v.join(' | '));
}

// B: kill at a random point (random pause hook), three runners started at the same instant.
async function B(E, i) {
  const pdf = E.pdf(`B${i}.pdf`, `B-${seedArg}-${i}`);
  const hook = pick([{}, { ENGINE_TEST_PAUSE_AFTER_RESERVE_MS: 1500 }, { ENGINE_TEST_PAUSE_AFTER_CACHE_MS: 1500 }, { ENGINE_TEST_PAUSE_AFTER_SETTLE_MS: 1500 }]);
  const target = pick(STAGES);
  setHooks(hook);
  let id; try { id = E.h.start_simulation({ pdf_path: pdf }).job_id; } finally { clearHooks(); }
  await E.waitFor(`state ${target}`, () => { const j = approveWhenAsked(E, id); return j.state === target || j.state === 'done'; }, 40000, 5);
  if (E.job(id).state === 'done') { rec(`B skipped (done early) ${target}`, true); return; }
  await sleep(Math.floor(rnd() * 900));
  if (pidAlive(E.runnerPid(id) || 0)) await killRunner(E, id);
  else { rec(`B runner already gone ${target}`, true); return; }
  await spawnMany(E, id, 3);
  await toDone(E, id); await settle(E, id);
  const v = check(E, id, { label: `B${i}[${target} ${JSON.stringify(hook)}]` });
  rec(`B kill@${target} hook=${Object.keys(hook)[0] || '-'} + 3 simultaneous runners`, v.length === 0, v.join(' | '));
}

// C: a takeover at a random point; the third runner is spawned in the same instant as the first is woken, and the second is killed
async function C_(E, i) {
  const pdf = E.pdf(`C${i}.pdf`, `C-${seedArg}-${i}`);
  const point = pick([...STAGES, ...PAID.map((p) => `${p}:reserved`), ...PAID.map((p) => `${p}:sent`)]);
  const f = await lib.frozenThenReplaced(E, { name: `C${i}.pdf`, point, secondEnv: pick([{}, { ENGINE_TEST_PAUSE_AFTER_RESERVE_MS: 800 }, { ENGINE_TEST_PAUSE_AFTER_CACHE_MS: 800 }]) });
  C.add('takeovers');
  const mode = pick(['wake-first', 'kill-second-then-wake', 'wake+spawn']);
  if (mode === 'wake-first') { f.wake(); await sleep(Math.floor(rnd() * 400)); }
  else if (mode === 'kill-second-then-wake') {
    await sleep(Math.floor(rnd() * 700));
    const p = E.runnerPid(f.id);
    if (p && p !== f.oldPid && pidAlive(p)) { process.kill(p); C.add('kills'); await sleep(30); }
    f.wake(); await spawnMany(E, f.id, 1); C.add('takeovers');
  } else { f.wake(); await spawnMany(E, f.id, 2); C.add('takeovers'); }
  await toDone(E, f.id, 70000);
  await lib.wakeAndLeave(E, f).catch(() => {}); // wake is idempotent
  await E.waitFor('first runner gone', () => !pidAlive(f.oldPid), 20000, 50).catch(() => {});
  await settle(E, f.id);
  const v = check(E, f.id, { label: `C${i}[${point} ${mode}]` });
  rec(`C takeover@${point} ${mode}`, v.length === 0, v.join(' | '));
}

// D: damage earlier outputs of a dead job (or an idle job) in different ways; rebuilt byte for byte, no later stage again, ledger does not grow
async function D(E, i) {
  const pdf = E.pdf(`D${i}.pdf`, `D-${seedArg}-${i}`);
  const target = pick(['packaging', 'qa']);
  let id = E.h.start_simulation({ pdf_path: pdf }).job_id;
  await E.waitFor('target', () => { const j = approveWhenAsked(E, id); return j.state === target; }, 40000, 5);
  await killRunner(E, id);
  const done = STAGES.slice(0, STAGES.indexOf(target));
  const rows0 = E.rows(id).length; const led0 = E.ledger(id).rows.length; const sends0 = PAID.map((p) => E.sends(id, p));
  const hashes = Object.fromEntries(done.map((st) => [st, sha(E.jobPath(id, `stages/${st}/stub.json`))]));
  const nd = 1 + Math.floor(rnd() * 3);
  const dmg = []; const pool = [...done];
  const cacheBefore = fs.readdirSync(E.jobPath(id, 'cache')).sort().join();
  for (let k = 0; k < nd && pool.length; k++) {
    const st = pool.splice(Math.floor(rnd() * pool.length), 1)[0];
    const f = E.jobPath(id, `stages/${st}/stub.json`);
    const how = pick(['delete', 'truncate', 'flip', 'append', 'empty']);
    if (how === 'delete') fs.rmSync(f);
    else if (how === 'truncate') fs.writeFileSync(f, fs.readFileSync(f).subarray(0, 10));
    else if (how === 'flip') { const b = fs.readFileSync(f); b[b.length - 3] ^= 1; fs.writeFileSync(f, b); }
    else if (how === 'append') fs.appendFileSync(f, ' ');
    else fs.writeFileSync(f, '');
    dmg.push(`${st}:${how}`);
  }
  E.h.start_simulation({ pdf_path: pdf });
  await toDone(E, id, 70000); await settle(E, id);
  const v = check(E, id, { label: `D${i}`, damaged: true });
  for (const st of done) if (sha(E.jobPath(id, `stages/${st}/stub.json`)) !== hashes[st]) v.push(`${st} not rebuilt byte for byte`);
  const man = E.rows(id);
  for (const st of STAGES) {
    const n = man.filter((r) => r.stage === st).length;
    const dm = dmg.some((d) => d.startsWith(st + ':'));
    if (dm && n !== 2) v.push(`damaged ${st} has ${n} rows (want 2)`);
    if (!dm && done.includes(st) && n !== 1) v.push(`undamaged finished stage ${st} ran again (${n} rows)`);
  }
  if (E.ledger(id).rows.length !== led0) v.push(`ledger grew ${led0} -> ${E.ledger(id).rows.length}`);
  PAID.forEach((p, k) => { if (E.sends(id, p) !== sends0[k]) v.push(`${p} sent again`); });
  if (fs.readdirSync(E.jobPath(id, 'cache')).sort().join() !== cacheBefore) v.push('cache files changed');
  rec(`D kill@${target} damage=${dmg.join(',')}`, v.length === 0, v.join(' | '));
}

// E: cancel during each pause of a paid call, then restart
async function E_(E, i) {
  const pdf = E.pdf(`E${i}.pdf`, `E-${seedArg}-${i}`);
  const hk = pick(['ENGINE_TEST_PAUSE_AFTER_RESERVE_MS', 'ENGINE_TEST_PAUSE_AFTER_CACHE_MS', 'ENGINE_TEST_PAUSE_AFTER_SETTLE_MS']);
  const stage = pick(PAID);
  setHooks({ [hk]: 2500 });
  let id; try { id = E.h.start_simulation({ pdf_path: pdf }).job_id; } finally { clearHooks(); }
  await E.waitFor('pause reached', () => { approveWhenAsked(E, id); return new RegExp(String.raw`test hook: pausing \d+ ms after the [a-z ]+ of seq ` + (PAID.indexOf(stage) + 1) + String.raw`\b`).test(E.log(id)) && E.job(id).state === stage; }, 40000, 10);
  const pid = E.runnerPid(id);
  E.h.cancel_job({ job_id: id });
  await E.waitFor('runner leaves', () => !pidAlive(pid), 15000, 30);
  const j = E.job(id);
  const man = E.rows(id).filter((r) => r.stage === stage);
  const led = E.ledger(id);
  const v = [];
  if (j.state !== 'cancelled') v.push('state ' + j.state);
  if (man.length !== 1) v.push(`${stage}: ${man.length} manifest rows after cancel in paid call (contract 10.6 wants 1)`);
  if (led.rows.filter((r) => r.stage === stage && r.status === 'settled').length !== 1) v.push('no single settled row');
  E.h.start_simulation({ pdf_path: pdf });
  await toDone(E, id, 70000); await settle(E, id);
  v.push(...check(E, id, { label: `E${i}` }));
  rec(`E cancel in ${hk.replace('ENGINE_TEST_PAUSE_AFTER_', '')} of ${stage}`, v.length === 0, v.join(' | '));
}

// F: locks left by DEAD processes never block a resume for more than about 2 s
async function F(E, i) {
  const dead = await new Promise((res) => { const c = spawn(process.execPath, ['-e', ''], { stdio: 'ignore' }); const p = c.pid; c.on('exit', () => res(p)); });
  const pdf = E.pdf(`F${i}.pdf`, `F-${seedArg}-${i}`);
  const target = pick(['tracing', 'building', 'rendering', 'skinning', 'gating']);
  let id = E.h.start_simulation({ pdf_path: pdf }).job_id;
  await E.waitFor('target', () => { const j = approveWhenAsked(E, id); return j.state === target; }, 40000, 5);
  await sleep(150);
  const killed = await killRunner(E, id);
  const what = pick(['job.lock', 'job.lock+pid', 'heavy.guard', 'heavy.lock', 'heavy.guard+heavy.lock+job.lock', 'job.lock-nl', 'job.lock-empty']);
  const heavy = E.config.runner.heavyLock;
  const t = new Date().toISOString();
  if (what.startsWith('job.lock') || what.includes('job.lock')) fs.writeFileSync(E.jobPath(id, 'job.lock'), what === 'job.lock-empty' ? '' : what === 'job.lock-nl' ? `${dead}\n` : String(dead));
  if (what.includes('heavy.guard')) { fs.mkdirSync(heavy.replace(/[^/\\]+$/, ''), { recursive: true }); fs.writeFileSync(heavy + '.guard', String(dead)); }
  if (what.includes('heavy.lock') && !what.endsWith('guard')) { fs.mkdirSync(heavy.replace(/[^/\\]+$/, ''), { recursive: true }); fs.writeFileSync(heavy, JSON.stringify({ pid: dead, job_id: id, at: t, refreshed_at: t })); }
  const t0 = Date.now();
  const r = E.h.start_simulation({ pdf_path: pdf });
  const claimed = await E.waitFor('new runner claims and beats', () => { const p = E.runnerPid(id); return p && p !== killed && pidAlive(p) && Date.now(); }, 15000, 5);
  const j0 = E.job(id);
  const prog = await E.waitFor('progress past the dead lock', () => { const j = approveWhenAsked(E, id); return (j.heartbeat_at !== j0.heartbeat_at || j.state !== target) && Date.now(); }, 15000, 5);
  const dt = prog - t0;
  await toDone(E, id, 70000); await settle(E, id);
  const v = check(E, id, { label: `F${i}` });
  if (dt > 3500) v.push(`resume blocked ${dt} ms by ${what}`);
  rec(`F dead-lock ${what} @${target} resumed=${r.resumed} first progress after ${dt} ms`, v.length === 0, v.join(' | '));
}
