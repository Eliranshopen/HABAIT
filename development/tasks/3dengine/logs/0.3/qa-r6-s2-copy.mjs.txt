// qa-r6 s2: a wrong copy of spent_usd in job.json in every state; every tool + both CLI commands; lock held; ledger damaged.
import fs from 'node:fs';
import { mkEnv, toDone, sleep, stats, say, cli, cliSpent, holdLock, killPid, ledgerSum, stubEngine, ledgerMod, atomic } from './common.mjs';

const E = await mkEnv({ prefix: 'qar6-s2-', staleSeconds: 2, heartbeat: 0.1, stepMs: 150 });
const bad = [];
const note = (m) => { say(m); };
try {
  const pdf = E.pdf('s2.pdf', 's2');
  const id = E.h.start_simulation({ pdf_path: pdf }).job_id;
  await toDone(E, id, 45000);
  await E.waitFor('runner gone', () => !E.runnerPid(id) || true, 1000);
  await sleep(1500);
  const truth = ledgerSum(E, id);
  say(`# s2 home=${E.home} job=${id} ledger settled=${truth}`);
  const h2 = stubEngine.createStubEngine(E.config, { doctor: { get: async () => null, kick() {}, stop() {} }, spawnRunner: () => {}, isRunnerAlive: () => false }).handlers;
  const jf = `${E.config.jobsDir}/${id}/job.json`;
  const snap = JSON.parse(fs.readFileSync(jf, 'utf8'));
  const states = {
    queued: (j) => { j.state = 'queued'; j.percent = 0; j.result = null; },
    skinning: (j) => { j.state = 'skinning'; j.percent = 70; j.result = null; },
    awaiting_approval: (j) => { j.state = 'awaiting_approval'; j.percent = 55; j.result = null; j.approval = null; j.quote = { estimate_usd: 2, range_usd: [1, 3], rooms: 1, basis: 'x' }; },
    cancelled: (j) => { j.state = 'cancelled'; j.result = null; },
    failed: (j) => { j.state = 'failed'; j.result = null; j.reason = { code: 'STAGE_FAILED', message: 'x' }; },
    review: (j) => { j.state = 'review'; j.result = null; },
    done: () => {},
  };
  const copies = [0, 99, 1.2, null, '1.8', 'unset', truth];
  const writeJob = (st, c) => {
    const j = structuredClone(snap);
    states[st](j);
    if (c === 'unset') delete j.cost.spent_usd; else j.cost.spent_usd = c;
    fs.writeFileSync(jf, JSON.stringify(j, null, 2));
  };
  const raw = () => JSON.parse(fs.readFileSync(jf, 'utf8')).cost.spent_usd;
  let asks = 0;
  const expect = (what, got) => { asks++; if (got !== truth) { bad.push(`${what}: reported ${JSON.stringify(got)} != ${truth}`); note(`  VIOLATION ${what}: ${JSON.stringify(got)}`); } };
  for (const st of Object.keys(states)) {
    for (const c of copies) {
      const tag = `${st}/copy=${JSON.stringify(c)}`;
      // each tool on a fresh wrong copy
      writeJob(st, c); expect(`${tag} get_job`, h2.get_job({ job_id: id }).job.cost.spent_usd);
      if (raw() !== truth) { bad.push(`${tag} get_job did not put the copy right (${raw()})`); note(`  VIOLATION ${tag} copy left ${raw()}`); }
      writeJob(st, c); expect(`${tag} list_jobs`, h2.list_jobs({ limit: 20 }).jobs.find((j) => j.job_id === id).cost.spent_usd);
      if (raw() !== truth) { bad.push(`${tag} list_jobs copy left ${raw()}`); }
      writeJob(st, c);
      try { const r = await h2.get_result({ job_id: id }); expect(`${tag} get_result`, r.cost.spent_usd); if (raw() !== truth) bad.push(`${tag} get_result copy left ${raw()}`); }
      catch (e) { if (!(st !== 'done' && /NOT_DONE|not "done"/.test(e.message + e.code))) { bad.push(`${tag} get_result threw ${e.message}`); } }
      writeJob(st, c);
      try { const r = h2.start_simulation({ pdf_path: pdf }); expect(`${tag} start_simulation`, r.job.cost.spent_usd); if (raw() !== truth) bad.push(`${tag} start_simulation copy left ${raw()}`); }
      catch (e) { bad.push(`${tag} start_simulation threw ${e.message}`); }
      writeJob(st, c);
      if (st === 'awaiting_approval') { try { const r = h2.approve_paid_stage({ job_id: id, cap_usd: 60 }); expect(`${tag} approve`, r.job.cost.spent_usd); if (raw() !== truth) bad.push(`${tag} approve copy left ${raw()}`); } catch (e) { bad.push(`${tag} approve threw ${e.message}`); } writeJob(st, c); }
      try { const r = h2.cancel_job({ job_id: id }); expect(`${tag} cancel_job`, r.job.cost.spent_usd); if (raw() !== truth) bad.push(`${tag} cancel copy left ${raw()}`); } catch (e) { bad.push(`${tag} cancel threw ${e.message}`); }
      // CLI (2 calls) on a fresh wrong copy, only for a subset to keep time
      if (c === 0 || c === 'unset' || c === null) {
        writeJob(st, c); const o = cliSpent(E, id);
        expect(`${tag} cli job --json`, o.cli_job); expect(`${tag} cli jobs --json`, o.cli_jobs);
        writeJob(st, c); try { cli(E, ['job', id]); } catch (e) { bad.push(`${tag} cli job (human) failed ${e.message.split('\n')[0]}`); }
      }
    }
  }
  say(`# s2 matrix: asks=${asks} violations=${bad.length}`);

  // lock held by a live process while we ask
  for (const [st, c, refresh, ms] of [['done', 0, false, 4000], ['skinning', 99, true, 5000], ['cancelled', null, false, 3000]]) {
    writeJob(st, c);
    const holder = await holdLock(`${E.config.jobsDir}/${id}/job.lock`, ms, { refresh });
    const t0 = Date.now();
    const g = h2.get_job({ job_id: id }).job.cost.spent_usd; const dt1 = Date.now() - t0;
    const t1 = Date.now();
    const l = h2.list_jobs({ limit: 20 }).jobs.find((j) => j.job_id === id).cost.spent_usd; const dt2 = Date.now() - t1;
    let gr = 'n/a';
    if (st === 'done') { try { gr = (await h2.get_result({ job_id: id })).cost.spent_usd; } catch (e) { gr = `ERR ${e.message}`; } }
    const copyDuring = raw();
    expect(`lockheld ${st} get_job`, g); expect(`lockheld ${st} list_jobs`, l); if (gr !== 'n/a') expect(`lockheld ${st} get_result`, gr);
    say(`  lock held ${ms}ms refresh=${refresh} ${st}: get_job=${g} (${dt1}ms) list_jobs=${l} (${dt2}ms) get_result=${gr} copy-during=${copyDuring}`);
    killPid(holder.pid); await sleep(300);
    try { fs.rmSync(`${E.config.jobsDir}/${id}/job.lock`, { force: true }); } catch { /* */ }
    const after = h2.get_job({ job_id: id }).job.cost.spent_usd; expect(`lockheld ${st} after`, after);
    if (raw() !== truth) { bad.push(`lock-held ${st}: copy not put right after the lock was free (${raw()})`); }
  }

  // damaged / removed ledger
  const lf = `${E.config.jobsDir}/${id}/ledger.json`;
  const lg = fs.readFileSync(lf, 'utf8');
  const dam = {
    deleted: () => fs.rmSync(lf),
    truncated: () => fs.writeFileSync(lf, lg.slice(0, Math.floor(lg.length / 2))),
    empty: () => fs.writeFileSync(lf, ''),
    array: () => fs.writeFileSync(lf, '[]'),
    rowsNull: () => fs.writeFileSync(lf, '{"rows":null}'),
    garbage: () => fs.writeFileSync(lf, 'not json'),
  };
  for (const [k, f] of Object.entries(dam)) {
    fs.writeFileSync(lf, lg); writeJob('done', 1.2); f();
    const res = {};
    const grab = async (n, fn) => { try { res[n] = await fn(); } catch (e) { res[n] = `ERR ${e.message.split('\n')[0]}`; } };
    await grab('get_job', () => h2.get_job({ job_id: id }).job.cost.spent_usd);
    await grab('list_jobs', () => h2.list_jobs({ limit: 20 }).jobs.find((j) => j.job_id === id)?.cost.spent_usd);
    await grab('get_result', async () => (await h2.get_result({ job_id: id })).cost.spent_usd);
    await grab('cli_job_json', () => JSON.parse(cli(E, ['job', id, '--json'])).cost.spent_usd);
    await grab('cli_job_human', () => { cli(E, ['job', id]); return 'ok'; });
    await grab('cli_ledger', () => { cli(E, ['ledger', id]); return 'ok'; });
    say(`  ledger ${k}: ${JSON.stringify(res)} copy-after=${raw()}`);
  }
  fs.writeFileSync(lf, lg);
  stats.violations.push(...bad);
  for (const b of bad) say(`  V: ${b}`);
} finally {
  await E.cleanup();
  say(`# s2 done violations=${bad.length} leftover=${JSON.stringify(E.leftover)}`);
}
