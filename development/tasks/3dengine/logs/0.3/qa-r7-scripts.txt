// Returns an array of violation strings (empty = fine).
export function checkJob(E, id, { final = false, rendering = null } = {}) {
  const v = [];
  const job = parse(E.jobPath(id, 'job.json'));
  const man = parse(E.jobPath(id, 'manifest.json'));
  const led = parse(E.jobPath(id, 'ledger.json'));
  if (!job) return ['job.json missing'];
  const rows = led?.rows || [];
  const done = job.state === 'done';
  const by = new Map();
  for (const r of rows) {
    if (!isAmt(r.reserved_usd)) v.push(`row ${r.seq}: reserved_usd not an amount (${r.reserved_usd})`);
    if (r.status === 'settled' && !isAmt(r.settled_usd)) v.push(`row ${r.seq}: settled row settled_usd not an amount (${r.settled_usd})`);
    if (r.status !== 'settled' && r.settled_usd !== null && !(r.status === 'failed' && r.settled_usd === 0) && !(r.status === 'orphaned' && r.settled_usd === null)) v.push(`row ${r.seq}: ${r.status} with settled_usd ${r.settled_usd}`);
    if (r.status === 'reserved' && done) v.push(`row ${r.seq}: reserved at done`);
    if (r.status === 'orphaned' && !r.sending_at) v.push(`row ${r.seq}: orphaned without sending_at`);
    if (r.not_sent && r.sending_at) v.push(`row ${r.seq}: not_sent with sending_at`);
    if (r.status === 'settled' && !r.sending_at) v.push(`row ${r.seq}: settled without sending_at`);
    if (r.status === 'settled') { const a = by.get(r.call_key) || []; a.push(r); by.set(r.call_key, a); }
    if (final && r.status === 'reserved') v.push(`row ${r.seq}: reserved at the end`);
  }
  for (const [k, a] of by) {
    const plain = a.filter((r) => !r.late && r.resend_of === undefined);
    if (plain.length > 1) v.push(`call ${k.slice(0, 8)}: ${plain.length} plain settled rows`);
  }
  // sends <= rows marked
  for (const s of PAID) {
    const sends = E.sends(id, s);
    const marked = rows.filter((r) => r.stage === s && r.sending_at).length;
    if (sends > marked) v.push(`${s}: ${sends} sends but ${marked} rows marked sending_at`);
  }
  const settled = Math.round(rows.filter((r) => r.status === 'settled').reduce((n, r) => n + (Number(r.settled_usd) || 0), 0) * 1e6) / 1e6;
  const held = Math.round(rows.filter((r) => r.status === 'reserved' || r.status === 'orphaned').reduce((n, r) => n + (Number(r.reserved_usd) || 0), 0) * 1e6) / 1e6;
  if (job.cost.spent_usd !== settled && (done || final)) v.push(`copy ${job.cost.spent_usd} != ledger ${settled}`);
  if (typeof led?.cap_usd === 'number' && settled + held > led.cap_usd + 1e-9) v.push(`counted ${settled + held} above cap ${led.cap_usd}`);
  // manifest: one valid (non superseded) done row per stage
  const mrows = man?.rows || [];
  for (const st of ['tracing', 'building', 'rendering', ...PAID, 'packaging', 'qa']) {
    const valid = mrows.filter((r) => r.stage === st && r.status === 'done' && !r.superseded);
    if (valid.length > 1) v.push(`stage ${st}: ${valid.length} non-superseded rows`);
    if (done && valid.length !== 1) v.push(`stage ${st}: ${valid.length} valid rows at done`);
  }
  if (fs.existsSync(E.jobPath(id, 'job.lock')) && final) v.push('job.lock left');
  const log = E.log(id);
  if (/\berror: /.test(log)) v.push(`runner.log has an error line: ${log.split('\n').find((l) => /\berror: /.test(l))}`);
  return v;
}

export function overlapRendering(E, ids) {
  const iv = [];
  for (const id of ids) for (const r of (parse(E.jobPath(id, 'manifest.json'))?.rows || [])) if (r.stage === 'rendering') iv.push({ id, a: Date.parse(r.started_at), b: Date.parse(r.finished_at) });
  const out = [];
  for (let i = 0; i < iv.length; i++) for (let j = i + 1; j < iv.length; j++) if (iv[i].a < iv[j].b && iv[j].a < iv[i].b) out.push(`rendering overlap ${iv[i].id} / ${iv[j].id}`);
  return out;
}

// What every report path says about spent_usd, vs the ledger.
export async function reports(E, id) {
  const led = parse(E.jobPath(id, 'ledger.json'));
  const truth = Math.round((led?.rows || []).filter((r) => r.status === 'settled').reduce((n, r) => n + (Number(r.settled_usd) || 0), 0) * 1e6) / 1e6;
  const bad = [];
  const g = E.h.get_job({ job_id: id }).job.cost.spent_usd;
  const l = E.h.list_jobs({ limit: 50 }).jobs.find((j) => j.job_id === id)?.cost.spent_usd;
  if (g !== truth) bad.push(`get_job ${g} != ${truth}`);
  if (l !== truth) bad.push(`list_jobs ${l} != ${truth}`);
  const j = E.job(id);
  if (j.state === 'done') {
    const r = await E.h.get_result({ job_id: id });
    if (r.cost.spent_usd !== truth) bad.push(`get_result ${r.cost.spent_usd} != ${truth}`);
  }
  return { truth, bad };
}

===== s4-chaos.mjs
// s4: kills and takeovers of my own design. node s4-chaos.mjs <type> <count> <seed>
//  chain  : runner1 frozen at a random point, runner2 takes over (random pause hook) and is KILLED, runner3 resumes, runner1 woken at a random moment
//  heavy  : two jobs, A frozen holding the heavy lock at rendering, B steals it, then kill B / kill A / wake A at random moments
//  rubble : runner killed at a random moment, then dead-pid / empty / garbage job.lock + heavy.lock.guard rubble planted, resume must start in about 2 s
//  pauses : two kills of two successive runners, each inside a (random) pause of a paid call; third runner finishes
//  cap    : a cap that stops the job at the second/third call, kills around the stop and the approval
//  cancel : frozen at paid:sent, cancel, restart (takeover) while blocked, wake
import fs from 'node:fs';
import { mkEnv, sleep, pidAlive, frozenJob, frozenThenReplaced, toDone, wakeAndLeave, checkJob, overlapRendering, reports } from './common.mjs';

const [type, countS, seedS] = process.argv.slice(2);
const COUNT = Number(countS || 5);
const box = { s: Number(seedS || 1) };
const rnd = () => { box.s = (box.s * 48271) % 2147483647; return box.s / 2147483647; };
const pick = (a) => a[Math.floor(rnd() * a.length)];
const between = (a, b) => a + Math.floor(rnd() * (b - a + 1));
const PAID = ['skinning', 'gating', 'finishing'];
const OUT = []; const say = (s) => { OUT.push(s); console.log(s); };
let kills = 0; let takeovers = 0; let violations = 0; let scenarios = 0;
const V = (tag, arr) => { for (const x of arr) { violations++; say(`VIOLATION ${tag}: ${x}`); } };

const E = await mkEnv({ prefix: `qa7-${type}-`, stepMs: 300, callUsd: type === 'cap' ? 25 : 0.6, heartbeat: 0.1, staleSeconds: 1 });
const kill = async (id) => { const p = await E.kill(id); kills++; return p; };
const finish = async (id, tag, { extra = [] } = {}) => {
  await toDone(E, id, 60000);
  await sleep(150);
  const v = checkJob(E, id, { final: true });
  const rp = await reports(E, id);
  V(tag, [...v, ...rp.bad, ...extra]);
  return rp.truth;
};
const waitLog = (id, text, n = 1, ms = 15000) => E.waitFor(`log ${text}`, () => E.log(id).split(text).length - 1 >= n, ms, 10);

async function chain(i) {
  const name = `chain${i}`;
  const paid = pick(PAID);
  const point = pick([paid, `${paid}:reserved`, `${paid}:sent`]);
  const hook = pick([{}, { ENGINE_TEST_PAUSE_AFTER_RESERVE_MS: 4000 }, { ENGINE_TEST_PAUSE_AFTER_CACHE_MS: 4000 }, { ENGINE_TEST_PAUSE_AFTER_SETTLE_MS: 4000 }]);
  const wakeAt = pick(['before3', 'after3', 'afterdone', 'afterkill']);
  const r = await frozenThenReplaced(E, { name, point, secondEnv: hook });
  takeovers++;
  const id = r.id;
  const hk = Object.keys(hook)[0];
  let desc = `${name} ${point} hook2=${hk || '-'} wake=${wakeAt}`;
  // kill runner2: inside its pause when it has one, else at a random moment
  if (hk) {
    try { await E.waitFor('pause line', () => E.log(id).includes('test hook: pausing'), 12000, 10); } catch { desc += ' (no pause reached)'; }
    await sleep(between(0, 300));
  } else await sleep(between(0, 700));
  let k2 = null;
  if (pidAlive(E.runnerPid(id)) && E.runnerPid(id) === r.newPid) { k2 = await kill(id); }
  if (wakeAt === 'afterkill') r.wake();
  if (wakeAt === 'before3') { r.wake(); await sleep(between(0, 200)); }
  const again = E.h.start_simulation({ pdf_path: r.pdf });
  if (again.resumed) takeovers += 0;
  if (wakeAt === 'after3') { await sleep(between(0, 500)); r.wake(); }
  // runner 3 may itself be killed (one in three)
  if (rnd() < 0.34) {
    await sleep(between(50, 900));
    const p3 = E.runnerPid(id);
    if (p3 && pidAlive(p3) && p3 !== r.pid) { await kill(id); E.h.start_simulation({ pdf_path: r.pdf }); desc += ' +k3'; }
  }
  await finish(id, desc);
  if (wakeAt === 'afterdone') { await sleep(100); r.wake(); }
  else r.wake();
  try { await E.waitFor('runner1 gone', () => !pidAlive(r.pid), 20000, 25); } catch { V(desc, ['replaced runner 1 did not leave by itself']); }
  // after it left: nothing changed by it
  await sleep(150);
  V(desc + ' (after runner1 left)', checkJob(E, id, { final: true }));
  takeovers++;   // runner1 woke to find itself replaced: a lock loss (ownership lost) counted once per scenario
  say(`ok ${desc} k2=${k2}`);
}

async function heavy(i) {
  const a = await frozenJob(E, { name: `hA${i}`, point: 'rendering' });
  await E.waitFor('A holds heavy', () => E.heavy()?.pid === a.pid, 10000, 20);
  const pdfB = E.pdf(`hB${i}`, `hB${i}`);
  const idB = E.h.start_simulation({ pdf_path: pdfB }).job_id;
  // B reaches rendering and (A being silent for > 1 s) steals the heavy lock
  await E.waitFor('B owns heavy', () => E.heavy()?.job_id === idB, 25000, 20);
  takeovers++;
  const mode = pick(['wakeA', 'killB', 'killA', 'wakeAafterB']);
  let desc = `heavy${i} ${mode}`;
  if (mode === 'wakeA') { await sleep(between(0, 150)); a.wake(); }
  if (mode === 'killB') { await sleep(between(0, 250)); const pb = E.runnerPid(idB); if (pb && pidAlive(pb)) { await kill(idB); E.h.start_simulation({ pdf_path: pdfB }); } a.wake(); }
  if (mode === 'killA') { process.kill(a.pid); await E.waitFor('A gone', () => !pidAlive(a.pid), 5000, 10); kills++; E.h.start_simulation({ pdf_path: a.pdf }); }
  if (mode === 'wakeAafterB') { await E.waitFor('B past rendering', () => E.rows(idB).some((r) => r.stage === 'rendering'), 20000, 20); a.wake(); }
  await finish(idB, `${desc} B`);
  await finish(a.id, `${desc} A`);
  if (mode !== 'killA') { try { await E.waitFor('A runner gone', () => !pidAlive(a.pid), 20000, 25); } catch { V(desc, ['A did not leave']); } }
  V(desc, overlapRendering(E, [a.id, idB]));
  const left = E.heavy();
  if (left) V(desc, [`heavy lock left behind: ${JSON.stringify(left)}`]);
  say(`ok ${desc} Aattempts(rendering)=${E.job(a.id).attempts?.rendering}`);
}

async function rubble(i) {
  const pdf = E.pdf(`rub${i}`, `rub${i}`);
  const id = E.h.start_simulation({ pdf_path: pdf }).job_id;
  const stagePick = pick(['tracing', 'building', 'rendering', 'awaiting', 'skinning', 'gating']);
  await E.waitFor('stage', () => { const j = E.job(id); if (j.state === 'awaiting_approval' && !j.approval && stagePick !== 'awaiting') E.approve(id); return stagePick === 'awaiting' ? j.state === 'awaiting_approval' : j.state === stagePick; }, 40000, 10);
  await sleep(between(0, 250));
  await kill(id);
  const kind = pick(['dead', 'deadSpaces', 'empty', 'garbage', 'zero', 'negative']);
  const lock = E.jobPath(id, 'job.lock');
  const content = { dead: '2147480001', deadSpaces: ' 2147480002\n', empty: '', garbage: 'abc', zero: '0', negative: '-7' }[kind];
  fs.writeFileSync(lock, content);
  const hl = `${E.config.runner.heavyLock}.guard`;
  const withHeavy = rnd() < 0.5;
  if (withHeavy) { fs.mkdirSync(hl.replace(/\/[^/]+$/, ''), { recursive: true }); fs.writeFileSync(hl, '2147480003'); }
  const t0 = Date.now();
  E.h.start_simulation({ pdf_path: pdf });   // may itself wait on the lock
  const tTool = Date.now() - t0;
  const hb0 = E.job(id).heartbeat_at;
  // time until a NEW live runner has really written something
  let tRun = null;
  try {
    await E.waitFor('new runner active', () => { const p = E.runnerPid(id); const j = E.job(id); const jl = fs.existsSync(lock) ? fs.readFileSync(lock, 'utf8') : null; return p && pidAlive(p) && j.heartbeat_at !== hb0 && jl !== content; }, 9000, 10);
    tRun = Date.now() - t0;
  } catch { tRun = 'never'; }
  const desc = `rubble${i} ${stagePick} ${kind} heavyGuard=${withHeavy}`;
  if (tRun === 'never' || tRun > 2600) V(desc, [`resume took ${tRun} ms (start_simulation call ${tTool} ms)`]);
  await finish(id, desc);
  if (fs.existsSync(hl)) { fs.rmSync(hl, { force: true }); }
  say(`ok ${desc} start=${tTool}ms active=${tRun}ms`);
}

async function pauses(i) {

  const hooks = ['ENGINE_TEST_PAUSE_AFTER_RESERVE_MS', 'ENGINE_TEST_PAUSE_AFTER_CACHE_MS', 'ENGINE_TEST_PAUSE_AFTER_SETTLE_MS'];
  const h1 = pick(hooks); const h2 = pick(hooks);
  process.env[h1] = '30000';
  const id = E.h.start_simulation({ pdf_path: pdf }).job_id;
  delete process.env[h1];
  await E.waitFor('pause 1', () => { const j = E.job(id); if (j.state === 'awaiting_approval' && !j.approval) E.approve(id); return E.log(id).includes('test hook: pausing'); }, 40000, 10);
  await sleep(between(0, 300));
  await kill(id);
  const n1 = E.log(id).split('test hook: pausing').length - 1;
  process.env[h2] = '30000';
  E.h.start_simulation({ pdf_path: pdf });
  delete process.env[h2];
  let got2 = true;
  try { await E.waitFor('pause 2 or done', () => { const j = E.job(id); if (j.state === 'awaiting_approval' && !j.approval) E.approve(id); return E.log(id).split('test hook: pausing').length - 1 > n1 || j.state === 'done'; }, 30000, 10); } catch { got2 = false; }
  if (E.job(id).state !== 'done') { await sleep(between(0, 300)); const p = E.runnerPid(id); if (p && pidAlive(p)) await kill(id); }
  E.h.start_simulation({ pdf_path: pdf });
  await finish(id, `pauses${i} ${h1.slice(-12)} then ${h2.slice(-12)}`);
  say(`ok pauses${i} ${h1.slice(-12)} then ${h2.slice(-12)} second pause reached=${got2}`);
}
