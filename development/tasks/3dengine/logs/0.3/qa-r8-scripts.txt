// bytes of every ledger file (and cache dirs listing) of a jobs folder, plus folder names
export function snapshot(E) {
  const out = { dirs: [], ledgers: {} };
  const jd = E.config.jobsDir;
  if (!fs.existsSync(jd)) return out;
  out.dirs = fs.readdirSync(jd).sort();
  for (const d of out.dirs) {
    const f = `${jd}/${d}/ledger.json`;
    try { out.ledgers[d] = sha(fs.readFileSync(f)); } catch { out.ledgers[d] = null; }
  }
  return out;
}
export const diffSnap = (a, b) => {
  const bad = [];
  if (JSON.stringify(a.dirs) !== JSON.stringify(b.dirs)) bad.push(`dirs ${a.dirs} -> ${b.dirs}`);
  for (const k of new Set([...Object.keys(a.ledgers), ...Object.keys(b.ledgers)])) if (a.ledgers[k] !== b.ledgers[k]) bad.push(`ledger ${k} changed`);
  return bad;
};

// Invariants from the files only. Returns list of violation strings.
export function check(E, id, { ended = true } = {}) {
  const v = [];
  const dir = `${E.config.jobsDir}/${id}`;
  const job = readJ(`${dir}/job.json`);
  const led = readJ(`${dir}/ledger.json`) || { rows: [] };
  const man = readJ(`${dir}/manifest.json`) || { rows: [] };
  if (!job) return ['no job.json'];
  const amt = (n) => typeof n === 'number' && Number.isFinite(n) && n >= 0 && n <= 1e6;
  let settled = 0;
  const byKey = {};
  led.rows.forEach((r, i) => {
    if (r.seq !== i + 1) v.push(`seq ${r.seq} at ${i}`);
    if (!amt(r.reserved_usd)) v.push(`bad reserved ${r.seq}`);
    if (r.status === 'settled') { if (!amt(r.settled_usd)) v.push(`bad settled ${r.seq}`); settled += r.settled_usd || 0; }
    if (r.status === 'orphaned' && !r.sending_at) v.push(`orphan without mark ${r.seq}`);
    if (job.state === 'done' && r.status === 'reserved') v.push(`reserved at done ${r.seq}`);
    (byKey[r.call_key] ||= []).push(r);
  });
  for (const rows of Object.values(byKey)) {
    const s = rows.filter((r) => r.status === 'settled');
    const plain = s.filter((r) => !r.late && r.resend_of === undefined);
    if (plain.length > 1) v.push(`${plain.length} plain settled rows for one call: ${plain.map((r) => r.seq)}`);
  }
  for (const st of ['skinning', 'gating', 'finishing']) {
    const sends = E.sends(id, st);
    const marked = led.rows.filter((r) => r.stage === st && r.sending_at).length;
    if (sends > marked) v.push(`${st}: ${sends} sends > ${marked} marked rows`);
  }
  settled = Math.round(settled * 1e6) / 1e6;
  if (job.cost.spent_usd !== settled && ended) v.push(`spent copy ${job.cost.spent_usd} != ledger ${settled}`);
  if (job.state === 'done') {
    for (const st of lib.FIXTURES ? jobsMod.STAGES : []) {
      const rows = man.rows.filter((r) => r.stage === st && r.status === 'done' && !r.superseded);
      if (rows.length !== 1) v.push(`${st}: ${rows.length} valid rows`);
      else if (!manifestMod.outputsIntact(dir, rows[0].outputs)) v.push(`${st}: outputs not intact`);
    }
  }
  if (ended && fs.existsSync(`${dir}/job.lock`)) v.push('job.lock left');
  return v;
}

// rows of a stage across two jobs overlapping
export function overlap(E, ids) {
  const iv = [];
  for (const id of ids) for (const r of (readJ(`${E.config.jobsDir}/${id}/manifest.json`)?.rows || [])) if (r.stage === 'rendering' && r.status === 'done') iv.push([Date.parse(r.started_at), Date.parse(r.finished_at), id, r.attempt]);
  const bad = [];
  for (let i = 0; i < iv.length; i++) for (let j = i + 1; j < iv.length; j++) if (iv[i][0] < iv[j][1] && iv[j][0] < iv[i][1]) bad.push(`rendering overlap ${iv[i]} ${iv[j]}`);
  return bad;
}

const AMT = (n) => typeof n === 'number' && Number.isFinite(n) && n >= 0 && n <= 1e6;
function specReads(text, id) {
  let d; try { d = JSON.parse(text); } catch { return false; }
  if (!d || typeof d !== 'object' || Array.isArray(d)) return false;
  if (d.schema !== 'habait/ledger.v1' || d.job_id !== id) return false;
  if (!(d.cap_usd === null || (typeof d.cap_usd === 'number' && Number.isFinite(d.cap_usd) && d.cap_usd >= 0))) return false;
  if (!Array.isArray(d.rows)) return false;
  for (let i = 0; i < d.rows.length; i++) {
    const r = d.rows[i];
    if (!r || typeof r !== 'object' || Array.isArray(r)) return false;
    if (r.seq !== i + 1) return false;
    if (typeof r.call_key !== 'string' || !r.call_key) return false;
    if (typeof r.stage !== 'string' || !r.stage) return false;
    if (!['reserved', 'settled', 'failed', 'orphaned'].includes(r.status)) return false;
    if (!AMT(r.reserved_usd)) return false;
    if (!(AMT(r.settled_usd) || (r.settled_usd === null && r.status !== 'settled'))) return false;
  }
  return true;
}

async function mkState(kind, name) {
  const pdf = E.pdf(`${name}.pdf`, name);
  if (kind === 'awaiting') {
    const id = E.h.start_simulation({ pdf_path: pdf }).job_id;
    await E.waitFor('awaiting', () => E.job(id).state === 'awaiting_approval', 30000);
    await sleep(200);
    try { await E.kill(id); } catch {}
    return { id, pdf };
  }
  if (kind === 'done') {
    const id = E.h.start_simulation({ pdf_path: pdf }).job_id;
    await lib.toDone(E, id, 60000);
    await sleep(300);
    return { id, pdf };
  }
  // skinning with a marked reserved row, runner killed inside the pause
  process.env.ENGINE_TEST_PAUSE_AFTER_RESERVE_MS = '60000';
  let id;
  try { id = E.h.start_simulation({ pdf_path: pdf }).job_id; } finally { delete process.env.ENGINE_TEST_PAUSE_AFTER_RESERVE_MS; }
  await E.waitFor('skinning reserved+marked', () => {
    const j = E.job(id);
    if (j.state === 'awaiting_approval' && !j.approval) E.approve(id);
    const l = readJ(E.jobPath(id, 'ledger.json'));
    return l && l.rows && l.rows.some((r) => r.sending_at && r.status === 'reserved');
  }, 60000);
  await E.kill(id);
  if (kind === 'cancelled') { E.h.cancel_job({ job_id: id }); }
  return { id, pdf };
}
