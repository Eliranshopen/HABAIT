async function round(kind, name, make) {
  // restore pristine, apply variant
  fs.writeFileSync(jx, PJ); fs.writeFileSync(mx, PM);
  if (kind === 'job') { const v = typeof make === 'string' ? make.slice(3) : JSON.stringify(make(pj())); fs.writeFileSync(jx, v); } else { const v = typeof make === 'string' ? make.slice(3) : JSON.stringify(make(pm())); fs.writeFileSync(mx, v); }

  const before = snapOthers(); const xBefore = snapshot(`${h.jobsDir}/${xid}`, (p) => /runner\.(log|pid)$/.test(p));
  const out = [];
  const r = runRunner(h, xid, 30000); n++; out.push(`runner exit ${r.status}${/stopped: unreadable/.test(r.log) ? ' (unreadable)' : ''}${r.log.includes(xid) ? ' names job' : ' NO JOB NAME'}`);
  for (const t of ['get_job', 'approve_paid_stage', 'cancel_job', 'get_result']) { n++; try { await H[t]({ job_id: xid, ...(t === 'approve_paid_stage' ? { cap_usd: 60 } : {}) }); out.push(`${t}:OK`); } catch (e) { out.push(`${t}:${e.code || e.name}`); } }
  n++; try { const s = await H.start_simulation({ pdf_path: X }); out.push(`start:created=${s.created}`); } catch (e) { out.push(`start:${e.code || e.name}`); }
  n++; try { const l = await H.list_jobs({ limit: 50 }); out.push(`list:${l.jobs.some((j) => j.job_id === xid) ? 'INCLUDES X' : 'omits X'}`); } catch (e) { out.push(`list:${e.code || e.name}`); }
  const c1 = cli(['job', xid]); n++; const c2 = cli(['jobs']); n++; const c3 = cli(['ledger', xid]); n++; const c4 = cli(['resume', xid]); n++;
  out.push(`cli job ${c1.status} jobs ${c2.status} ledger ${c3.status} resume ${c4.status}`);
  // wait for any runner the CLI resume may have spawned to finish / kill it
  await sleep(500);
  const rp = runnerPid(h, xid); if (rp && pidAlive(rp)) killPid(rp);
  await sleep(300);
  const after = snapOthers(); const xAfter = snapshot(`${h.jobsDir}/${xid}`, (p) => /runner\.(log|pid)$/.test(p));
  const changedOthers = diffSnap(before, after);
  const xChanged = diffSnap(xBefore, xAfter);
  const undefinedDir = fs.existsSync(`${h.jobsDir}/undefined`) || fs.existsSync(`${h.jobsDir}/null`);
  const top = topLevel() !== topBase;
  const sameFiles = fs.readFileSync(mx, 'utf8') === poisonedM && (kind === 'manifest' || fs.readFileSync(jx, 'utf8') === poisonedJ);  // job.json may be changed by approve/cancel when ITS file is valid
  const looks = out.join(' | ');
  const okX = xChanged.filter((k) => k !== 'job.json' && k !== 'manifest.json').length === 0;
  const ok = changedOthers.length === 0 && !undefinedDir && !top && sameFiles;
  if (!ok) bad.push(`${kind}/${name}`);
  table.push(`${ok ? 'ok ' : 'BAD'} ${kind} ${name}: ${looks}${changedOthers.length ? ' OTHERS CHANGED ' + changedOthers.join(',') : ''}${undefinedDir ? ' UNDEFINED-DIR' : ''}${top ? ' NEW-TOPLEVEL ' + topLevel() : ''}${sameFiles ? '' : ' X-FILES-REWRITTEN'}${okX ? '' : ' X-OTHER-FILES ' + xChanged.join(',')}`);
}
