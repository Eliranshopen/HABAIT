// D2: every way a look cannot be stored, through the real MCP server (stdio), both tools.
import fs from 'node:fs';
import path from 'node:path';
import { spawn } from 'node:child_process';
import { makeHome, startServer, E, SCR, GOLDEN, sha, cleanup, sleep } from './qa-lib.mjs';
const S = sha(fs.readFileSync(GOLDEN));
const FIXED = "the first look could not be stored in the engine's runtime folder";
const results = [];

async function scenario(name, setup, { during } = {}) {
  for (const tool of ['inspect_plan', 'start_simulation']) {
    const h = await makeHome({ timeout: 60 });
    const pdf = `${h.inbox}/plan.pdf`;
    fs.copyFileSync(GOLDEN, pdf);
    const srv = startServer(h);
    await srv.init();
    const rt = h.config.runtimeDir;
    await setup({ h, rt, srv });
    let killer;
    if (during) killer = await during({ h, rt, srv });
    const r = await srv.call(tool, { pdf_path: pdf });
    if (killer) killer.kill();
    const e = r.err; const m = r.m;
    let verdict;
    if (m.error) verdict = `RPC-ERROR ${m.error.code} ${m.error.message}`;
    else if (e) verdict = `tool-error ${e.code}: ${e.message}`;
    else verdict = `OK (answered) ${Object.keys(r.out).slice(0, 4)}`;
    const raw = JSON.stringify(m);
    const leak = /ENOENT|ENOTDIR|EEXIST|EPERM|EACCES|EISDIR|ENOTEMPTY|[A-Z]:[\\/]|qa2-home/.test(raw);
    const jobsMade = fs.existsSync(h.jobsDir) ? fs.readdirSync(h.jobsDir).filter((n) => !n.startsWith('.')) : [];
    const fixed = e?.code === 'TOOLCHAIN_MISSING' && e.message === FIXED;
    results.push({ name, tool, verdict, fixedMsg: fixed, leakOfPathOrRawCode: leak, jobsMade: jobsMade.length });
    console.log(JSON.stringify(results[results.length - 1]));
    await srv.stop();
    cleanup(h);
  }
}
