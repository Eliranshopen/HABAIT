function killAll(h) {
  // kill runners of this home only
  const r = spawnSync('powershell', ['-NoProfile', '-Command', `Get-CimInstance Win32_Process | Where-Object { $_.CommandLine -like '*runner.mjs*' -and $_.CommandLine -like '*${h.home.split('/').pop()}*' } | ForEach-Object { $_.ProcessId }`], { encoding: 'utf8', windowsHide: true });
  void r;
  // runner command lines do not carry the home (it is in the env); use runner.pid files
  try { for (const d of fs.readdirSync(h.jobsDir)) { try { const p = Number(fs.readFileSync(`${h.jobsDir}/${d}/runner.pid`, 'utf8')); if (p) spawnSync('taskkill', ['/PID', String(p), '/T', '/F'], { windowsHide: true }); } catch {} } } catch {}
}

const summary = [];
async function callAll(srv, h, label, withJob) {
  const out = [];
  const seq = [['inspect_plan', { pdf_path: `${h.inbox}/s.pdf` }], ['start_simulation', { pdf_path: `${h.inbox}/s.pdf` }],
    ['get_job', { job_id: ID }], ['approve_paid_stage', { job_id: ID, cap_usd: 100 }], ['get_result', { job_id: ID }], ['list_jobs', {}], ['cancel_job', { job_id: ID }], ['start_simulation', { pdf_path: `${h.inbox}/s.pdf` }]];
  for (const [name, args] of seq) {
    let line;
    try {
      const r = await srv.call(name, args);
      if (r.m.error) line = { kind: 'RPC', msg: r.m.error.message };
      else if (r.err) line = { kind: 'TOOL', code: r.err.code, msg: r.err.message };
      else line = { kind: 'ok' };
    } catch (e) { line = { kind: 'CLIENT', msg: e.message }; }
    const flag = (line.kind === 'RPC' && PATHISH.test(line.msg)) ? ' !!PATH-IN-RPC' : (line.kind === 'TOOL' && PATHISH.test(line.msg) && line.code !== 'TOOLCHAIN_MISSING' ? ' ??path-in-tool' : '');
    out.push(`${name}: ${line.kind}${line.code ? ' ' + line.code : ''}${line.msg ? ' | ' + line.msg.slice(0, 140) : ''}${flag}`);
    if (flag.startsWith(' !!')) summary.push(`DEFECT? ${label} ${name}: ${line.msg}`);
    if (line.kind === 'CLIENT') summary.push(`CLIENT-ERR ${label} ${name}: ${line.msg}`);
  }
  return out;
}
