function makeHome(trace) {
  const home = fs.mkdtempSync(path.join(os.tmpdir(), 'qa12-home-')).replace(/\\/g, '/');
  fs.writeFileSync(`${home}/engine.config.json`, JSON.stringify({ ...BASE, python: REAL.python, pipelineRoot: REAL.pipelineRoot, appDepsDir: REAL.appDepsDir,
    legalDir: REAL.legalDir, requirements: REAL.requirements, envRoot: '.', ...(trace ? { trace } : {}) }));
  return home;
}
function procs() {
  const r = spawnSync('powershell', ['-NoProfile', '-Command',
    "Get-CimInstance Win32_Process | Where-Object { $_.CommandLine -like '*qa12-*' -or $_.CommandLine -like '*m12-home-*' } | ForEach-Object { \"$($_.ProcessId) $($_.Name) $($_.CommandLine.Substring(0, [Math]::Min(110, $_.CommandLine.Length)))\" }"], { encoding: 'utf8' });
  return (r.stdout || '').split(/\r?\n/).filter((l) => l.trim() && !/Get-CimInstance/.test(l));
}
const cleanEnv = () => { const e = { ...process.env }; for (const k of Object.keys(e)) if (/^(FAL_KEY|OPENROUTER_API_KEY)$/i.test(k)) delete e[k]; return e; };
function cli(home, args, extraEnv = {}) {
  return new Promise((resolve) => {
    const t0 = Date.now();
    const c = spawn(process.execPath, [`${HOME}/engine/cli.mjs`, ...args], { env: { ...cleanEnv(), ENGINE_HOME: home, ...extraEnv }, cwd: HOME, windowsHide: true });
    let out = '', err = ''; c.stdout.on('data', (d) => (out += d)); c.stderr.on('data', (d) => (err += d));
    c.on('exit', (code) => resolve({ code, out, err, secs: ((Date.now() - t0) / 1000).toFixed(1), pid: c.pid }));
  });
}
