const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
const out = [];
const say = (...a) => { const s = a.join(' '); out.push(s); console.log(s); };
const readJ = (f) => { try { return JSON.parse(fs.readFileSync(f, 'utf8')); } catch { return null; } };
function walk(dir, skip) { const res = []; for (const n of fs.readdirSync(dir)) { const p = `${dir}/${n}`; if (skip(p)) continue; const st = fs.statSync(p); if (st.isDirectory()) res.push(...walk(p, skip)); else res.push(p); } return res; }
function leak(h, ids) {
  const hits = [];
  if (h.lines.join('\n').includes(MARK)) hits.push('tool answers');
  for (const id of ids) {
    const d = h.jobDir(id);
    for (const f of walk(d, (p) => /\/stages\/tracing\/attempt-\d+$/.test(p) || /\/input$/.test(p))) {
      const buf = fs.readFileSync(f); if (buf.includes(MARK)) hits.push(f.replace(d, ''));
    }
  }
  return hits;
}
const ENDED = ['done', 'review', 'failed', 'cancelled', 'awaiting_approval'];
let RD = null;
function setReading(h, m) { const d = RD || h.readingsDir; fs.mkdirSync(d, { recursive: true }); fs.writeFileSync(`${d}/${GOLDEN_SHA}.json`, JSON.stringify(muts[m]())); }
async function runOne(h, label, id) {
  const t0 = Date.now();
  const { job, seen } = await h.follow(id, ENDED, 480000);
  await h.runnersGone();
  const d = h.jobDir(id);
  const man = readJ(`${d}/manifest.json`); const led = readJ(`${d}/ledger.json`);
  const j = readJ(`${d}/job.json`);
  say(`[${label}] state=${job.state} reason=${job.reason?.code} | ${job.reason?.message} | review=${JSON.stringify(j.review ?? null)} attempts=${JSON.stringify(j.attempts)} manifestRows=${man?.rows?.length} ledgerRows=${led?.rows?.length} spent=${job.cost?.spent_usd} stagesDir=${JSON.stringify(fs.readdirSync(`${d}/stages`))} tracingDirs=${JSON.stringify(fs.readdirSync(`${d}/stages/tracing`))} seen=${seen.join('>')} secs=${Math.round((Date.now() - t0) / 1000)}`);
  return job;
}
