def read(rel):
    return io.open(ROOT + rel, encoding='utf-8', newline='').read()


def patch(rel, pairs, limit=None):
    raw = read(rel)
    nl = '\r\n' if '\r\n' in raw else '\n'
    t = raw.replace('\r\n', '\n')
    for a, b in pairs:
        if b in t:
            continue
        if t.count(a) != 1:
            raise SystemExit('%s: %d matches for %r' % (rel, t.count(a), a[:70]))
        t = t.replace(a, b)
    if limit:
        for i, ln in enumerate(t.split('\n'), 1):
            if len(ln) > limit and '3dengine' in ln.lower():
                print('  OVER %d: %s line %d = %d chars' % (limit, rel, i, len(ln)))
    io.open(ROOT + rel, 'w', encoding='utf-8', newline='').write(t.replace('\n', nl))
    print('patched', rel)


def line_of(rel, start, optional=False):
    hits = [ln for ln in read(rel).replace('\r\n', '\n').split('\n') if ln.startswith(start)]
    if optional and not hits:
        return None
    if len(hits) != 1:
        raise SystemExit('%s: %d lines start with %r' % (rel, len(hits), start[:60]))
    return hits[0]
