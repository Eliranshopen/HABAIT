def patch(name, pairs):
    p = ROOT + name
    raw = io.open(p, encoding='utf-8', newline='').read()
    nl = '\r\n' if '\r\n' in raw else '\n'
    t = raw.replace('\r\n', '\n')
    for a, b in pairs:
        if b in t:
            continue
        if t.count(a) != 1:
            sys.exit('%s: found %d times: %s' % (name, t.count(a), a[:80]))
        t = t.replace(a, b)
    if name.endswith('CLAUDE.md'):
        for i, line in enumerate(t.split('\n'), 1):
            if len(line) > 600:
                sys.exit('%s line %d is %d characters' % (name, i, len(line)))
    io.open(p, 'w', encoding='utf-8', newline='').write(t.replace('\n', nl))
    print('patched', name)


def line_of(name, start):
    t = io.open(ROOT + name, encoding='utf-8', newline='').read().replace('\r\n', '\n')
    hits = [ln for ln in t.split('\n') if ln.startswith(start)]
    if len(hits) != 1:
        sys.exit('%s: %d lines start with %r' % (name, len(hits), start[:60]))
    return hits[0]


def line_or(name, start, new):
    t = io.open(ROOT + name, encoding='utf-8', newline='').read().replace('\r\n', '\n')
    if new in t:
        return new
    return line_of(name, start)
