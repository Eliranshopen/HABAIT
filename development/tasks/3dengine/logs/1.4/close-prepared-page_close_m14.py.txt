import io
import re
import sys

DAY = sys.argv[1] if len(sys.argv) > 1 else '18.11.2026'

raw = io.open(P, encoding='utf-8', newline='').read()
nl = '\r\n' if '\r\n' in raw else '\n'
t = raw.replace('\r\n', '\n')


def swap(old, new, what):
    global t
    if new in t and old not in t:
        return
    if t.count(old) != 1:
        raise SystemExit('%d matches for %s' % (t.count(old), what))
    t = t.replace(old, new)


def line_starting(start):
    hits = [ln for ln in t.split('\n') if ln.lstrip().startswith(start)]
    if len(hits) != 1:
        raise SystemExit('%d lines start with %r' % (len(hits), start[:50]))
    return hits[0]


# The milestone after 2.1, as the page names it: the first milestone item after m2-1.
m = re.search(r'<li class="m[^"]*" id="m2-1">.*?<li class="m[^"]*" id="(m\d+-\d+)">\s*<span class="m-no">([\d.]+)</span>.*?<div class="m-top"><b>(.*?)</b>', t, re.S)
if not m:
    raise SystemExit('the milestone after 2.1 was not found on the page')

print('the milestone after 2.1 on the page:', NEXT_NO, NEXT_NAME)

# ---- the eyebrow and the strip at the top
t = re.sub(r'מעודכן \d{2}\.\d{2}\.\d{4}</span>', 'מעודכן %s</span>' % DAY, t, count=1)

     'ריצה שנייה לא משלמת ונותנת תכנית זהה בית-לבית. שלב הפענוח הושלם. עכשיו <bdi>2.1</bdi> מפורקת למשימות: <bdi>Blender</bdi> מופעל לראשונה בתוך המנוע, קודם על חדר אחד.</p></div>', 'now item')
nxt = line_starting('<div class="now-item"><span class="st new">אחריה</span>')
swap(nxt,
     '    <div class="now-item"><span class="st new">אחריה</span><p><b><bdi>%s</bdi> %s.</b></p></div>' % (NEXT_NO, NEXT_NAME), 'next item')

swap('aria-label="4 אבני דרך: 3 סגורות, 1 בעבודה, 0 טרם התחילו"><i class="done" title="1.1 קליטה והצצה · סגורה"></i><i class="done" title="1.2 פענוח 846 במנוע · סגורה"></i><i class="done" title="1.3 שער הפענוח · סגורה"></i><i class="cur" title="1.4 קריאה שמורה · בעבודה"></i>',
     'aria-label="4 אבני דרך: 4 סגורות, 0 בעבודה, 0 טרם התחילו"><i class="done" title="1.1 קליטה והצצה · סגורה"></i><i class="done" title="1.2 פענוח 846 במנוע · סגורה"></i><i class="done" title="1.3 שער הפענוח · סגורה"></i><i class="done" title="1.4 קריאה שמורה · סגורה"></i>',
     'stage 1 cells')
