const { loadConfig } = await import(pathToFileURL(`${ENGINE}/engine/config.mjs`).href);
const fwd = (p) => p.split(String.fromCharCode(92)).join('/');
export const freePort = () => new Promise((res, rej) => {
  const s = net.createServer();
  s.once('error', rej);
  s.listen(0, '127.0.0.1', () => { const p = s.address().port; s.close(() => res(p)); });
});
export async function makeHome(tag) {
  const REAL = loadConfig({ env: {} });
  const BASE = JSON.parse(fs.readFileSync(`${ENGINE}/engine.config.json`, 'utf8'));
  const home = fwd(fs.mkdtempSync(path.join(os.tmpdir(), `q3home-${tag}-`)));
  fs.mkdirSync(`${home}/uploads`, { recursive: true });
  fs.mkdirSync(`${home}/site/tour`, { recursive: true });
  fs.writeFileSync(`${home}/site/tour/index.html`, '<p>stub</p>');
  fs.writeFileSync(`${home}/engine.config.json`, JSON.stringify({ ...BASE, python: REAL.python, pipelineRoot: REAL.pipelineRoot, appDepsDir: REAL.appDepsDir,
    legalDir: REAL.legalDir, requirements: REAL.requirements, envRoot: '.', inbox: [`${home}/uploads`], stub: { ...BASE.stub, siteRoot: `${home}/site` },
    runner: { ...BASE.runner, heartbeatSeconds: 0.5 }, inspect: { ...BASE.inspect, timeoutSeconds: 120 }, mode: 'real', trace: { ...BASE.trace, rounds: 1 } }));
  const port = await freePort();
  if (port === 8891) throw new Error('port 8891');
  const config = loadConfig({ env: { ENGINE_HOME: home } });
  return { home, config, port, jobs: `${home}/jobs-temp` };
}
export function envFor(h, extra = {}) {
  const env = { ...process.env, ENGINE_HOME: h.home, ENGINE_JOBS_DIR: h.jobs, ENGINE_SERVE_PORT: String(h.port), ...extra };
  for (const k of Object.keys(env)) if (/^(fal_key|openrouter_api_key)$/i.test(k)) delete env[k];
  env.OPENROUTER_API_KEY = FAKE_KEY;
  return env;
}

});
export const requests = (dir) => (fs.existsSync(`${dir}/requests.jsonl`) ? fs.readFileSync(`${dir}/requests.jsonl`, 'utf8').trim().split('\n').filter(Boolean).length : 0);
export function cli(h, args, { env = {}, timeout = 120000 } = {}) {
  const r = spawnSync(process.execPath, ['engine/cli.mjs', ...args], { cwd: ENGINE, encoding: 'utf8', timeout, windowsHide: true, maxBuffer: 32 << 20, env: envFor(h, env) });
  return { status: r.status, out: r.stdout || '', err: r.stderr || '', error: r.error?.message };
}
